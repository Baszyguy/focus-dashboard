    // Fetch Data Function (Supports both Google Apps Script doGet JSON & Demo Data fallback)
    async function fetchData() {
      const spinner = document.getElementById('refresh-spinner');
      if (spinner) spinner.classList.add('fa-spin');

      setConnectionStatus('syncing', 'SYNCING...');

      const savedUrl = localStorage.getItem('gas_telemetry_url');
      const targetUrl = savedUrl || APPS_SCRIPT_URL;

      // Check if URL is invalid or in demo mode
      if (isDemoMode || !targetUrl || targetUrl.includes("YOUR_APPS_SCRIPT_ID_HERE")) {
        setTimeout(() => {
          rawData = [...DEMO_DATA];
          processData();
          setConnectionStatus('demo', 'DEMO MODE');
          if (spinner) spinner.classList.remove('fa-spin');
        }, 300);
        return;
      }

      try {
        const response = await fetch(targetUrl, {
          method: 'GET',
          redirect: 'follow'
        });

        if (!response.ok) {
          throw new Error(`HTTP Status Error: ${response.status}`);
        }

        const data = await response.json();

        if (Array.isArray(data)) {
          rawData = data;
          processData();
          setConnectionStatus('online', 'LIVE ONLINE');
          showToast(`Synced ${data.length} telemetry records from Google Sheet`, 'success');
        } else {
          throw new Error('Returned data is not a valid JSON array');
        }
      } catch (error) {
        console.error('Fetch Error:', error);
        setConnectionStatus('offline', 'OFFLINE / ERROR');
        showToast('Sync Failed! Check URL & CORS settings. Displaying fallback data.', 'error');
        rawData = [...DEMO_DATA];
        processData();
      } finally {
        if (spinner) spinner.classList.remove('fa-spin');
      }
    }

    // Calculate KPI Totals matching Spreadsheet Formulas
    function calculateKPIs() {
      const totalRounds = rawData.length;

      // Count Unique Players
      const uniquePlayersSet = new Set(rawData.map(r => String(r.playerID || '').trim()).filter(Boolean));
      const totalUniquePlayers = uniquePlayersSet.size;

      let sumReaction = 0;
      let sumLevel = 0;
      let sumFalseAlarms = 0;
      let quitEarlyCount = 0;

      rawData.forEach(item => {
        sumReaction += Number(item.avgReactionTime) || 0;
        sumLevel += Number(item.maxLevel) || 0;
        sumFalseAlarms += Number(item.falseAlarms) || 0;

        const stat = String(item.status || '').toLowerCase();
        if (stat.includes('quit') || stat.includes('ยกเลิก')) {
          quitEarlyCount++;
        }
      });

      const avgReaction = totalRounds > 0 ? (sumReaction / totalRounds).toFixed(1) : 0;
      const avgLevel = totalRounds > 0 ? (sumLevel / totalRounds).toFixed(1) : 0;

      // Update DOM
      document.getElementById('kpi-total-rounds').textContent = totalRounds.toLocaleString();
      document.getElementById('kpi-total-players').textContent = totalUniquePlayers.toLocaleString();
      document.getElementById('kpi-avg-reaction').innerHTML = `${avgReaction}<span class="text-xs font-normal text-amber-500">ms</span>`;
      document.getElementById('kpi-avg-level').textContent = avgLevel;
      document.getElementById('kpi-false-alarms').textContent = sumFalseAlarms.toLocaleString();
      document.getElementById('kpi-quit-early').textContent = quitEarlyCount.toLocaleString();
    }

    // Render Left Leaderboard (Sorted by highest score)
    function renderLeaderboard() {
      const container = document.getElementById('leaderboard-container');
      const search = document.getElementById('leaderboard-search').value.toLowerCase().trim();

      if (!rawData || rawData.length === 0) {
        container.innerHTML = `
          <div class="text-center py-10 text-emerald-800 font-mono text-xs">
            <i class="fa-solid fa-ghost text-2xl mb-2"></i>
            <p>No player telemetry records found.</p>
          </div>
        `;
        document.getElementById('leaderboard-count-badge').textContent = `0 PLAYERS`;
        return;
      }

      // Group by PlayerID to extract each player's highest score performance (matching doGet logic)
      const playerMap = {};
      rawData.forEach(row => {
        const pId = String(row.playerID || 'UNKNOWN').trim();
        const score = Number(row.totalScore) || 0;

        if (!playerMap[pId] || score > (playerMap[pId]._numericScore || 0)) {
          playerMap[pId] = {
            ...row,
            _numericScore: score
          };
        }
      });

      let leaderboardArray = Object.values(playerMap);

      // Sort by score descending
      leaderboardArray.sort((a, b) => (Number(b.totalScore) || 0) - (Number(a.totalScore) || 0));

      // Filter by search term
      if (search) {
        leaderboardArray = leaderboardArray.filter(item => 
          String(item.playerName || '').toLowerCase().includes(search) ||
          String(item.playerID || '').toLowerCase().includes(search)
        );
      }

      document.getElementById('leaderboard-count-badge').textContent = `${leaderboardArray.length} PLAYERS`;

      if (leaderboardArray.length === 0) {
        container.innerHTML = `<div class="text-center py-8 text-emerald-700 font-mono text-xs">No matching players found.</div>`;
        return;
      }

      let html = '';
      leaderboardArray.forEach((player, index) => {
        const rank = index + 1;
        
        let rankBadge = `<span class="w-6 h-6 rounded flex items-center justify-center font-mono font-bold text-xs bg-emerald-950 text-emerald-400 border border-emerald-800">#${rank}</span>`;
        let borderStyle = "border-emerald-900/40 hover:border-emerald-500/50";

          borderStyle = "border-amber-500/40 bg-gradient-to-r from-amber-950/20 to-transparent";
          glowScore = "neon-text-gold";
        } else if (rank === 2) {
          rankBadge = `<span class="w-6 h-6 rounded flex items-center justify-center font-mono font-bold text-xs bg-slate-400/20 text-slate-200 border border-slate-400/50">2</span>`;
          borderStyle = "border-slate-500/30";
        } else if (rank === 3) {
          rankBadge = `<span class="w-6 h-6 rounded flex items-center justify-center font-mono font-bold text-xs bg-amber-800/30 text-amber-500 border border-amber-800/50">3</span>`;
          borderStyle = "border-amber-800/30";
        }

        html += `
          <div class="p-3 rounded-lg bg-[#080e0a] border ${borderStyle} transition duration-200 flex items-center justify-between gap-3">
            <div class="flex items-center gap-2.5 min-w-0">
              ${rankBadge}
              <div class="truncate">
                <div class="font-bold text-xs text-white truncate flex items-center gap-1.5">
                  <span>${escapeHtml(player.playerName || 'Unknown')}</span>
                  <span class="text-[10px] text-emerald-600 font-mono font-normal">(${escapeHtml(player.playerID || '')})</span>
                </div>
                <div class="text-[10px] text-emerald-600/80 font-mono flex items-center gap-2 mt-0.5">
                  <span><i class="fa-solid fa-bolt text-amber-500/80"></i> ${player.avgReactionTime || '0'}ms</span>
                  <span>•</span>
                  <span>Lvl ${player.maxLevel || '0'}</span>

    // Render Detailed Raw Data Table
    function renderRawTable() {
      const tbody = document.getElementById('raw-table-body');
      const search = document.getElementById('raw-table-search').value.toLowerCase().trim();
      const statusFilter = document.getElementById('status-filter').value;

      let list = [...rawData];

      // Filter Status
      if (statusFilter !== 'ALL') {
        if (statusFilter === 'Quit Early') {
          list = list.filter(item => String(item.status || '').toLowerCase().includes('quit') || String(item.status || '').includes('ยกเลิก'));
        } else {
          list = list.filter(item => !String(item.status || '').toLowerCase().includes('quit') && !String(item.status || '').includes('ยกเลิก'));
        }
      }

      // Filter Search
      if (search) {
        list = list.filter(item => 
          String(item.playerID || '').toLowerCase().includes(search) ||
          String(item.playerName || '').toLowerCase().includes(search) ||
          String(item.status || '').toLowerCase().includes(search)
        );
      }

      // Dynamic Column Sorting
      list.sort((a, b) => {
        let valA = a[sortColumn];
        let valB = b[sortColumn];

        if (!isNaN(Number(valA)) && !isNaN(Number(valB))) {
          valA = Number(valA);
          valB = Number(valB);
        } else {
          valA = String(valA || '').toLowerCase();
          valB = String(valB || '').toLowerCase();
        }

        if (valA < valB) return sortAscending ? -1 : 1;
        if (valA > valB) return sortAscending ? 1 : -1;
        return 0;
      });

      document.getElementById('raw-table-count').textContent = `Showing ${list.length} of ${rawData.length} records`;

      if (list.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="8" class="text-center py-8 text-emerald-800 font-mono">
              No matching records in raw dataset.
            </td>
          </tr>
        `;
        return;
      }

      let html = '';
      list.forEach(item => {
        const statusStr = String(item.status || 'Active');
        const isQuit = statusStr.toLowerCase().includes('quit') || statusStr.includes('ยกเลิก');

        const statusBadge = isQuit
          ? `<span class="px-2 py-0.5 rounded text-[10px] bg-amber-950/80 text-amber-400 border border-amber-700/50"><i class="fa-solid fa-triangle-exclamation"></i> Quit Early</span>`

        html += `
          <tr class="hover:bg-emerald-950/30 transition">
            <td class="p-3 font-semibold text-emerald-300">${escapeHtml(item.playerID || '-')}</td>
            <td class="p-3 text-white font-medium">${escapeHtml(item.playerName || 'Unknown')}</td>
            <td class="p-3 text-emerald-600 text-[11px]">${escapeHtml(item.timestamp || '-')}</td>
            <td class="p-3 text-center text-purple-300 font-bold">${item.maxLevel || '0'}</td>
            <td class="p-3 text-right text-amber-300">${item.avgReactionTime || '0'} ms</td>
            <td class="p-3 text-right text-emerald-400 font-bold">${Number(item.totalScore || 0).toLocaleString()}</td>
            <td class="p-3 text-center text-rose-400">${item.falseAlarms || '0'}</td>
            <td class="p-3 text-center">${statusBadge}</td>
          </tr>
        `;
      });

      tbody.innerHTML = html;
    }

    function sortTable(column) {
      if (sortColumn === column) {
        sortAscending = !sortAscending;
      } else {
        sortColumn = column;
        sortAscending = false;
      }
      renderRawTable();
    }

    function updateCharts() {
      if (!performanceChart || !statusChart) return;

      const playerMap = {};
      rawData.forEach(row => {
        const name = row.playerName || row.playerID || 'Player';
        const score = Number(row.totalScore) || 0;
        const react = Number(row.avgReactionTime) || 0;

        if (!playerMap[name] || score > playerMap[name].score) {
          playerMap[name] = { score, react };
        }
      });

      const topPlayers = Object.entries(playerMap)
        .sort((a, b) => b[1].score - a[1].score)
        .slice(0, 8);

      performanceChart.data.labels = topPlayers.map(p => p[0]);
      performanceChart.data.datasets[0].data = topPlayers.map(p => p[1].score);
      performanceChart.data.datasets[1].data = topPlayers.map(p => p[1].react);
      performanceChart.update();

      let quitCount = 0;
      let completeCount = 0;

      rawData.forEach(item => {
        const stat = String(item.status || '').toLowerCase();
        if (stat.includes('quit') || stat.includes('ยกเลิก')) {
          quitCount++;
        } else {
          completeCount++;
        }
      });

      statusChart.data.datasets[0].data = [completeCount, quitCount];
      statusChart.update();

      const legendBox = document.getElementById('status-legend');
      legendBox.innerHTML = `
        <div class="flex items-center gap-1.5">
          <span class="w-2.5 h-2.5 rounded-full bg-emerald-500 inline-block"></span>
          <span>Completed: <strong class="text-white">${completeCount}</strong></span>
        </div>
        <div class="flex items-center gap-1.5">
          <span class="w-2.5 h-2.5 rounded-full bg-amber-500 inline-block"></span>
          <span>Quit Early: <strong class="text-white">${quitCount}</strong></span>
        </div>
      `;
    }

    function toggleAutoRefresh() {
      autoRefreshEnabled = !autoRefreshEnabled;
      const statusText = document.getElementById('auto-status-text');

      if (autoRefreshEnabled) {
        statusText.textContent = "ON";
        statusText.className = "text-emerald-300";

    function startAutoRefresh() {
      stopAutoRefresh();
      autoRefreshInterval = setInterval(() => {
        fetchData();
      }, 10000);
    }

    function stopAutoRefresh() {
      if (autoRefreshInterval) {
        clearInterval(autoRefreshInterval);
        autoRefreshInterval = null;
      }
    }

    // HTML Escaping Utility
    function escapeHtml(str) {
      return String(str).replace(/[&<>"']/g, function(m) {
        return {
          '&': '&amp;',
          '<': '&lt;',
          '>': '&gt;',
          '"': '&quot;',
          "'": '&#039;'
        }[m];
      });
    }
